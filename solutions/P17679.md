## 暴力 dp

设 $t$ 为字母的种类数，本题中 $t = 26$。同时记 $L = \sum|s_i|$。

考虑设 $f_{i, c}$ 表示已经正确输入了 $i$ 个单词，且第 $i$ 个单词的最后一个字母为 $c$ 时的最小代价。

转移的衔接只取决于上一个单词的末字母与当前单词的首字母之间距离和新单词内部的花费：

$$f_{i, b} =\min_{end_l = b} \min_{c}\Bigl(f_{i-1,\, c} + cost_l + dis_{c,\, start_l}\Bigr)$$

其中 $l$ 枚举当前选择的单词，$c$ 枚举上一个单词的末字母（$0 \le c < 26$），$start_l / end_l$ 为单词 $l$ 的首/末字母，$cost_l$ 为单词 $l$ 内部从首字母依次走到末字母的代价。时间复杂度 $O(L + T^2 + knt)$。

::::warning[暴力代码]

```cpp
#include <bits/stdc++.h>
using namespace std;
int n, k;
vector<char> s[100005];
long long cost[100005];
vector<char> l1, l2, l31, l32;
int dis[30][30];
pair<int, int> rev[30];
void get(vector<char> &res)
{
    res.clear();
    char c = getchar_unlocked();
    while (c < 'a' || c > 'z')
        c = getchar_unlocked();
    while ('a' <= c && c <= 'z')
        res.push_back(c), c = getchar_unlocked();
}
int gdis(int a, int b)
{
    auto &pa = rev[a], &pb = rev[b];
    return abs(pa.first - pb.first) + abs(pa.second - pb.second);
}
void init()
{
    for (int i = 0; i < 9; ++i)
        rev[l1[i] - 'a'] = {1, i + 1};
    for (int i = 0; i < 9; ++i)
        rev[l2[i] - 'a'] = {2, i + 1};
    for (int i = 0; i < 4; ++i)
        rev[l31[i] - 'a'] = {3, i + 1};
    for (int i = 0; i < 4; ++i)
        rev[l32[i] - 'a'] = {3, i + 6};
    for (int i = 0; i < 26; ++i)
        for (int j = 0; j < 26; ++j)
            dis[i][j] = gdis(i, j);
    for (int i = 1; i <= n; ++i)
        for (int j = 1; j < (int)s[i].size(); ++j)
            cost[i] += gdis(s[i][j] - 'a', s[i][j - 1] - 'a');
}
const long long INF = 0x7f7f7f7f7f7f7f7fll;
long long dp[2][27];

int main()
{
    scanf("%d%d", &n, &k);
    for (int i = 1; i <= n; ++i)
        get(s[i]);
    get(l1);
    get(l2);
    get(l31);
    get(l32);
    init();
    int cur = 0;
    for (int i = 0; i < 26; ++i)
        dp[cur][i] = 0;
    for (int i = 1; i <= k; ++i)
    {
        int nxt = cur ^ 1;
        for (int x = 0; x < 26; ++x)
            dp[nxt][x] = INF;
        for (int j = 1; j <= n; ++j)
        {
            int fc = s[j][0] - 'a';
            int lc = s[j][s[j].size() - 1] - 'a';
            long long best = INF;
            for (int kk = 0; kk < 26; ++kk)
                best = min(best, dp[cur][kk] + gdis(kk, fc));
            dp[nxt][lc] = min(dp[nxt][lc], best + cost[j]);
        }
        cur = nxt;
    }

    long long ans = INF;
    for (int i = 0; i < 26; ++i)
        ans = min(ans, dp[cur][i]);
    printf("%lld\n", ans);
    return 0;
}
```

## 正解

注意到一个状态的后效性只体现在最后停留在字母的位置上，因此我们其实不关心实际上选的是哪个单词，故可以考虑从字母 $a$ 到字母 $b$ 的最小代价，仿照暴力 dp 的方案计算：

$$M_{a, b} = \min_{end_l = b}(cost_l + dis(a, start_l))$$

因此上述的 dp 转移可以简化为：

$$f_{i, b} = \min (f_{i - 1, c} + M_{c, b})$$

这个转移可以用 $(\min, +)$ 矩阵快速幂解决。最终答案为 $\min(M^k)$。

时间复杂度 $O(L + t^2 + nt + t^3 \log k)$。

### 关于 $(\min, +)$ 矩阵

普通的矩阵乘法是：

$$
C_{i,j}=\sum_t A_{i,t}B_{t,j}
$$

而 min-plus 矩阵将 $\Sigma$ 变为 $\min$，将乘法变为加法，即：

$$
(A \times B)_{i, j}=\min_t(A_{i, t} + B_{t, j})
$$

其单位矩阵为：

$$
I_{i, j} = \begin{cases}
0, & i = j \\
+\infty, &i\ne j
\end{cases}
$$

#### 结合律证明

令 $A, B, C$ 为三个 min-plus 矩阵。

要证 $(A\times B)\times C = A \times(B\times C)$

将左边根据定义拆开得：

$$
\begin{aligned}
((A\times B)\times C)_{i, j} = \min_t((A\times B)_{i, t} + C_{t, j}) &= \min_t\left(\min_l(A_{i, l} + B_{l, t}) + C_{t, j}\right) \\
&= \min_{t, l} (A_{i, l} + B_{l, t} + C_{t, j})
\end{aligned}
$$

右侧同理。

因此 min-plus 矩阵满足结合律，可以适用矩阵快速幂。

## 程序

```cpp
#include <bits/stdc++.h>
using namespace std;
int n;
long long k;
vector<char> s[100005];
long long cost[100005];
vector<char> l1, l2, l31, l32;
int dis[30][30];
pair<int, int> rev[30];
void get(vector<char> &res)
{
    res.clear();
    char c = getchar_unlocked();
    while (c < 'a' || c > 'z')
        c = getchar_unlocked();
    while ('a' <= c && c <= 'z')
        res.push_back(c), c = getchar_unlocked();
}
int gdis(int a, int b)
{
    auto &pa = rev[a], &pb = rev[b];
    return abs(pa.first - pb.first) + abs(pa.second - pb.second);
}
void init()
{
    for (int i = 0; i < 9; ++i)
        rev[l1[i] - 'a'] = {1, i + 1};
    for (int i = 0; i < 9; ++i)
        rev[l2[i] - 'a'] = {2, i + 1};
    for (int i = 0; i < 4; ++i)
        rev[l31[i] - 'a'] = {3, i + 1};
    for (int i = 0; i < 4; ++i)
        rev[l32[i] - 'a'] = {3, i + 6};
    for (int i = 0; i < 26; ++i)
        for (int j = 0; j < 26; ++j)
            dis[i][j] = gdis(i, j);
    for (int i = 1; i <= n; ++i)
        for (int j = 1; j < (int)s[i].size(); ++j)
            cost[i] += gdis(s[i][j] - 'a', s[i][j - 1] - 'a');
}
const long long INF = 0x7f7f7f7f7f7f7f7fll;
struct Mat
{
    long long a[26][26];
    Mat operator*(const Mat &B) {
        Mat C;
        for (int i = 0; i < 26; ++i)
            for (int j = 0; j < 26; ++j)
                C.a[i][j] = INF;

        for (int i = 0; i < 26; ++i)
        {
            for (int t = 0; t < 26; ++t)
            {
                if (a[i][t] >= INF / 2)
                    continue;

                for (int j = 0; j < 26; ++j)
                {
                    if (B.a[t][j] >= INF / 2)
                        continue;

                    long long val = a[i][t] + B.a[t][j];
                    if (val < C.a[i][j])
                        C.a[i][j] = val;
                }
            }
        }
        return C;
    }
};
int main()
{
    scanf("%d%lld", &n, &k);
    for (int i = 1; i <= n; ++i)
        get(s[i]);
    get(l1);
    get(l2);
    get(l31);
    get(l32);
    init();
    Mat M;
    for (int i = 0; i < 26; ++i)
        for (int j = 0; j < 26; ++j)
            M.a[i][j] = INF;
    for (int j = 1; j <= n; ++j)
    {
        int fc = s[j][0] - 'a';
        int lc = s[j][s[j].size() - 1] - 'a';
        for (int c = 0; c < 26; ++c)
        {
            long long val = cost[j] + dis[c][fc];
            if (val < M.a[c][lc])
                M.a[c][lc] = val;
        }
    }
    Mat res; // 初始为单位矩阵
    for (int i = 0; i < 26; ++i)
        for (int j = 0; j < 26; ++j)
            res.a[i][j] = (i == j ? 0 : INF);
    long long exp = k;
    while (exp > 0)
    {
        if (exp & 1LL)
            res = res * M;
        M = M * M;
        exp >>= 1LL;
    }

    long long ans = INF;
    for (int i = 0; i < 26; ++i)
        for (int j = 0; j < 26; ++j)
            ans = min(ans, res.a[i][j]);
    printf("%lld\n", ans);
    return 0;
}
```

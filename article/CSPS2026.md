附 diff.sh（linux，注意对拍代码均无需开文件读写）：

```sh
#!/bin/bash

g++ sol1.cpp -o sol1 -O2 -std=c++14 -static
g++ sol2.cpp -o sol2 -O2 -std=c++14 -static
g++ gen.cpp -o gen -O3 -std=c++14 -static
for i in {1..10000}; do
    gen > in
    sol1 < in > a
    sol2 < in > b
    if ! diff -q a b > /dev/null; then
        echo wa
        cp in wa.in
        cp a wa1.out
        cp b wa2.out
        break
    fi
    echo success $i
done
```
